select * from employe where employe.nemp NOT IN ( SELECT nemp from TRANSPORT );


SELECT C.NOMCH, E.NOMEMP
FROM CHANTIER C
JOIN DEPLACEMENT D ON C.NCH = D.NCH
JOIN TRANSPORT T ON D.NDEP = T.NDEP
JOIN EMPLOYE E ON T.NEMP = E.NEMP
WHERE D.DTDEP = TO_DATE('2024-02-02', 'YYYY-MM-DD');



SELECT E.NOMEMP, SUM(D.KMARRIVEE - D.KMDEPART) AS KM_TOTAL
FROM EMPLOYE E
JOIN TRANSPORT T ON E.NEMP = T.NEMP
JOIN DEPLACEMENT D ON T.NDEP = D.NDEP
GROUP BY E.NOMEMP;
